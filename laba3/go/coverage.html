// Array - динамический массив строк.
type Array struct {
        data     []string
        size     int
        capacity int
}

// NewArray создает пустой массив с начальной вместимостью 4.
func NewArray() *Array {
        capacity := 4
        return &Array{data: make([]string, capacity), size: 0, capacity: capacity}
}

// resizeArray увеличивает вместимость массива в два раза.
func (arr *Array) resizeArray() {
        newCapacity := arr.capacity * 2
        newData := make([]string, newCapacity)
        for i := 0; i < arr.size; i++ {
                newData[i] = arr.data[i]
        }
        arr.data = newData
        arr.capacity = newCapacity
}

// PushBack добавляет элемент в конец.
func (arr *Array) PushBack(text string) {
        if arr.size >= arr.capacity {
                arr.resizeArray()
        }
        arr.data[arr.size] = text
        arr.size++
}

// InsertAt вставляет элемент на место index.
func (arr *Array) InsertAt(text string, index int) {
        if index < 0 || index > arr.size {
                fmt.Println("такого индекса нету")
                return
        }

        if arr.size == arr.capacity {
                arr.resizeArray()
        }

        // сдвигаем элементы вправо, начиная с последнего,
        // чтобы не затереть соседа справа
        for i := arr.size - 1; i >= index; i-- {
                arr.data[i+1] = arr.data[i]
        }

        arr.data[index] = text
        arr.size++
}

// RemoveAt удаляет элемент с места index.
func (arr *Array) RemoveAt(index int) {
        if index < 0 || index >= arr.size {
                fmt.Println("такого индекса нету")
                return
        }

        // сдвигаем элементы влево, удаляемый затирается первым
        for i := index; i < arr.size-1; i++ {
                arr.data[i] = arr.data[i+1]
        }
        arr.size--
}

// SetAt заменяет элемент на месте index.
func (arr *Array) SetAt(text string, index int) {
        if index < 0 || index >= arr.size {
                fmt.Println("такого индекса нету")
                return
        }
        arr.data[index] = text
}

// GetAt возвращает элемент с места index.
// При неверном индексе возвращает пустую строку.
func (arr *Array) GetAt(index int) string {
        if index < 0 || index >= arr.size {
                fmt.Println("Нет такого индекса")
                return ""
        }
        return arr.data[index]
}

// PrintArray печатает все элементы, размер и вместимость.
func (arr *Array) PrintArray() {
        for i := 0; i < arr.size; i++ {
                fmt.Print(arr.data[i], " ")
        }
        fmt.Printf("(size=%d, capacity=%d)\n", arr.size, arr.capacity)
}

// Length возвращает количество элементов.
func (arr *Array) Length() int {
        return arr.size
}

// HandleArray выполняет команду над массивом с именем name.
func HandleArray(command Command, db DB, name string, ss io.Reader) {
        if WrongType(db, name, "array") {
                return
        }

        arr := NewArray()
        for _, item := range loadStrings(db, name) {
                arr.PushBack(item)
        }

        var index int
        var value string
        switch command {
        case MPUSH:
                fmt.Fscan(ss, &value)
                arr.PushBack(value)
                fmt.Println("->", value)
        case MINSERT:
                fmt.Fscan(ss, &index, &value)
                arr.InsertAt(value, index)
        case MGET:
                fmt.Fscan(ss, &index)
                fmt.Println("->", arr.GetAt(index))
        case MSET:
                fmt.Fscan(ss, &index, &value)
                arr.SetAt(value, index)
        case MDEL:
                fmt.Fscan(ss, &index)
                arr.RemoveAt(index)
        case MLEN:
                fmt.Println("->", arr.Length())
        case MPRINT:
                arr.PrintArray()
        }

        items := make([]string, 0, arr.Length())
        for i := 0; i < arr.Length(); i++ {
                items = append(items, arr.GetAt(i))
        }
        saveStrings(db, name, "array", items)
}

// SaveBinary сохраняет массив в бинарный файл:
// количество элементов, затем строки.
func (arr *Array) SaveBinary(fileName string) {
        out, err := os.Create(fileName)
        if err != nil {
                return
        }
        defer out.Close()

        writeInt(out, arr.size)
        for i := 0; i < arr.size; i++ {
                writeString(out, arr.data[i])
        }
}

// LoadBinary загружает массив из бинарного файла.
// Старые элементы перед загрузкой удаляются.
func (arr *Array) LoadBinary(fileName string) {
        in, err := os.Open(fileName)
        if err != nil {
                return
        }
        defer in.Close()

        arr.size = 0
        count := readInt(in)
        for i := 0; i < count; i++ {
                arr.PushBack(readString(in))
        }
}

// ToCommand превращает текст команды в Command.

                return command

        return UNKNOWN

// Entry - одна структура в файле базы: ее тип и данные.
type Entry struct {
        Type string          `json:"type"`
        Data json.RawMessage `json:"data"`
}

// DB - вся база: имя структуры и ее запись.
type DB map[string]*Entry

// WrongType проверяет, что под этим именем не лежит структура другого типа.
func WrongType(db DB, name, typ string) bool {
        entry, ok := db[name]
        if ok && entry.Type != typ {
                fmt.Println(name, "это не", typ)
                return true
        }
        return false
}

// loadStrings достает из базы данные структуры, которая хранит строки.
func loadStrings(db DB, name string) []string {
        var items []string
        if entry, ok := db[name]; ok {
                _ = json.Unmarshal(entry.Data, &items)
        }
        return items
}

// loadInts достает из базы данные структуры, которая хранит числа.
func loadInts(db DB, name string) []int {
        var items []int
        if entry, ok := db[name]; ok {
                _ = json.Unmarshal(entry.Data, &items)
        }
        return items
}

        if items == nil {
                items = []string{}
        }

        db[name] = &Entry{Type: typ, Data: data}
}

}

import (

// DNode - узел двусвязного списка.
type DNode struct {
        Person string
        Next   *DNode
        Prev   *DNode
}

// Dlist - двусвязный список с головой и хвостом.
type Dlist struct {
        head *DNode
        tail *DNode
}

// NewDlist создает пустой список.
func NewDlist() *Dlist {
        return &Dlist{}
}

// IsEmptyList проверяет, пуст ли список.
func (dl *Dlist) IsEmptyList() bool {
        return dl.head == nil && dl.tail == nil
}

// PushBack добавляет элемент в конец.
func (dl *Dlist) PushBack(name string) {
        newNode := &DNode{Person: name}

        if dl.IsEmptyList() {
                dl.head = newNode
                dl.tail = newNode
        } else {
                dl.tail.Next = newNode
                newNode.Prev = dl.tail
                dl.tail = newNode
        }
}

// PopBack удаляет последний элемент.
func (dl *Dlist) PopBack() {
        if dl.IsEmptyList() {
                return
        }

        dl.tail = dl.tail.Prev
        if dl.tail == nil {
                // узел был единственным, список опустел
                dl.head = nil
        } else {
                dl.tail.Next = nil
        }
}

// PushFront добавляет элемент в начало.
func (dl *Dlist) PushFront(name string) {
        newNode := &DNode{Person: name}

        if dl.IsEmptyList() {
                dl.head = newNode
                dl.tail = newNode
        } else {
                dl.head.Prev = newNode
                newNode.Next = dl.head
                dl.head = newNode
        }
}

// PopFront удаляет первый элемент.
func (dl *Dlist) PopFront() {
        if dl.IsEmptyList() {
                return
        }

        dl.head = dl.head.Next
        if dl.head == nil {
                // узел был единственным, список опустел
                dl.tail = nil
        } else {
                dl.head.Prev = nil
        }
}

// PushBetween вставляет элемент после узла node.
func (dl *Dlist) PushBetween(node *DNode, name string) {
        if node == nil {
                return
        }

        newNode := &DNode{Person: name, Next: node.Next, Prev: node}
        if node.Next != nil {
                node.Next.Prev = newNode
        } else {
                dl.tail = newNode
        }
        node.Next = newNode
}

// PushBefore вставляет элемент перед узлом node.
func (dl *Dlist) PushBefore(node *DNode, name string) {
        if node == nil {
                return
        }

        if node.Prev == nil {
                dl.PushFront(name)
        } else {
                dl.PushBetween(node.Prev, name)
        }
}

// PopAfter удаляет элемент после узла node.
func (dl *Dlist) PopAfter(node *DNode) {
        if node == nil || node.Next == nil {
                return
        }

        tmp := node.Next
        if tmp.Next == nil {
                // удаляем хвост
                node.Next = nil
                dl.tail = node
                return
        }

        node.Next = tmp.Next
        tmp.Next.Prev = node
}

// PopBefore удаляет элемент перед узлом node.
func (dl *Dlist) PopBefore(node *DNode) {
        if node == nil || node.Prev == nil {
                return
        }

        tmp := node.Prev
        if tmp.Prev == nil {
                // удаляем голову
                node.Prev = nil
                dl.head = node
                return
        }

        tmp.Prev.Next = node
        node.Prev = tmp.Prev
}

// PrintForward печатает список от головы к хвосту.
func (dl *Dlist) PrintForward() {
        for current := dl.head; current != nil; current = current.Next {
                fmt.Print(current.Person, " <-> ")
        }
        fmt.Println("nil")
}

// PrintBackward печатает список от хвоста к голове.
func (dl *Dlist) PrintBackward() {
        for current := dl.tail; current != nil; current = current.Prev {
                fmt.Print(current.Person, " <-> ")
        }
        fmt.Println("nil")
}

// FindName ищет узел со значением name, если не нашел - возвращает nil.
func (dl *Dlist) FindName(name string) *DNode {
        for current := dl.head; current != nil; current = current.Next {
                if current.Person == name {
                        return current
                }
        }
        return nil
}

// PopName удаляет первый элемент со значением name.
func (dl *Dlist) PopName(name string) {
        node := dl.FindName(name)
        if node == nil {
                fmt.Println("нет такого элемента")
                return
        }

        if node == dl.head {
                dl.PopFront()
        } else {
                dl.PopAfter(node.Prev)
        }
}

// GetHead возвращает первый узел списка.
func (dl *Dlist) GetHead() *DNode {
        return dl.head
}

// HandleDlist выполняет команду над двусвязным списком с именем name.
func HandleDlist(command Command, db DB, name string, ss io.Reader) {
        if WrongType(db, name, "dlist") {
                return
        }

        dl := NewDlist()
        for _, item := range loadStrings(db, name) {
                dl.PushBack(item)
        }

        var target, value string
        var node *DNode
        switch command {
        case LPUSHH:
                fmt.Fscan(ss, &value)
                dl.PushFront(value)
        case LPUSHT:
                fmt.Fscan(ss, &value)
                dl.PushBack(value)
        case LPUSHA:
                fmt.Fscan(ss, &target, &value)
                node = dl.FindName(target)
                if node == nil {
                        fmt.Println("нет такого элемента")
                } else {
                        dl.PushBetween(node, value)
                }
        case LPUSHB:
                fmt.Fscan(ss, &target, &value)
                node = dl.FindName(target)
                if node == nil {
                        fmt.Println("нет такого элемента")
                } else {
                        dl.PushBefore(node, value)
                }
        case LDELH:
                dl.PopFront()
        case LDELT:
                dl.PopBack()
        case LDELA:
                fmt.Fscan(ss, &target)
                node = dl.FindName(target)
                if node == nil {
                        fmt.Println("нет такого элемента")
                } else {
                        dl.PopAfter(node)
                }
        case LDELB:
                fmt.Fscan(ss, &target)
                node = dl.FindName(target)
                if node == nil {
                        fmt.Println("нет такого элемента")
                } else {
                        dl.PopBefore(node)
                }
        case LDEL:
                fmt.Fscan(ss, &value)
                dl.PopName(value)
        case LFIND:
                fmt.Fscan(ss, &value)
                if dl.FindName(value) != nil {
                        fmt.Println("-> TRUE")
                } else {
                        fmt.Println("-> FALSE")
                }
        case LPRINT:
                dl.PrintForward()
        case LPRINTR:
                dl.PrintBackward()
        }

        items := []string{}
        for current := dl.GetHead(); current != nil; current = current.Next {
                items = append(items, current.Person)
        }
        saveStrings(db, name, "dlist", items)
}

// SaveBinary сохраняет список в бинарный файл:

import (

// FNode - узел односвязного списка.
type FNode struct {
        Value string
        Next  *FNode
}

// Flist - односвязный список.
type Flist struct {
        head *FNode
}

// NewFlist создает пустой список.
func NewFlist() *Flist {
        return &Flist{}
}

// IsEmptyList проверяет, пуст ли список.
func (fl *Flist) IsEmptyList() bool {
        return fl.head == nil
}

// PushFront добавляет элемент в начало.
func (fl *Flist) PushFront(a string) {
        fl.head = &FNode{Value: a, Next: fl.head}
}

// PopFront удаляет первый элемент.
func (fl *Flist) PopFront() {
        if fl.IsEmptyList() {
                return
        }
        fl.head = fl.head.Next
}

// AddOne вставляет элемент после узла first.
func (fl *Flist) AddOne(first *FNode, a string) {
        if first == nil {
                return
        }
        first.Next = &FNode{Value: a, Next: first.Next}
}

// DeleteNode удаляет элемент после узла abc.
func (fl *Flist) DeleteNode(abc *FNode) {
        if abc == nil || abc.Next == nil {
                return
        }
        abc.Next = abc.Next.Next
}

// AddTail добавляет элемент в конец.
// Хвоста у списка нет, поэтому идем до конца от головы.
func (fl *Flist) AddTail(value string) {
        newNode := &FNode{Value: value}

        if fl.IsEmptyList() {
                fl.head = newNode
                return
        }

        tmp := fl.head
        for tmp.Next != nil {
                tmp = tmp.Next
        }
        tmp.Next = newNode
}

// PopTail удаляет последний элемент.
func (fl *Flist) PopTail() {
        if fl.IsEmptyList() {
                return
        }

        if fl.head.Next == nil {
                fl.head = nil
                return
        }

        tmp := fl.head
        for tmp.Next.Next != nil {
                tmp = tmp.Next
        }
        tmp.Next = nil
}

// AddBefore вставляет элемент перед узлом target.
func (fl *Flist) AddBefore(target *FNode, a string) {
        if target == nil {
                return
        }

        if target == fl.head {
                fl.PushFront(a)
                return
        }

        p := fl.head
        for p.Next != target {
                p = p.Next
        }
        fl.AddOne(p, a)
}

// DeleteBefore удаляет элемент перед узлом target.
func (fl *Flist) DeleteBefore(target *FNode) {
        if fl.IsEmptyList() || target == nil || target == fl.head {
                return
        }

        if fl.head.Next == target {
                fl.PopFront()
                return
        }

        p := fl.head
        for p.Next.Next != target {
                p = p.Next
        }
        fl.DeleteNode(p)
}

// FindValue ищет узел со значением a, если не нашел - возвращает nil.
func (fl *Flist) FindValue(a string) *FNode {
        for current := fl.head; current != nil; current = current.Next {
                if current.Value == a {
                        return current
                }
        }
        return nil
}

// DeleteValue удаляет первый элемент со значением a.
func (fl *Flist) DeleteValue(a string) {
        if fl.IsEmptyList() {
                return
        }

        if fl.head.Value == a {
                fl.PopFront()
                return
        }

        p := fl.head
        for p.Next != nil && p.Next.Value != a {
                p = p.Next
        }

        if p.Next != nil {
                fl.DeleteNode(p)
        }
}

// printReverseFrom рекурсивно печатает список с конца, начиная с node.
func (fl *Flist) printReverseFrom(node *FNode) {
        if node == nil {
                return
        }
        fl.printReverseFrom(node.Next)
        fmt.Print(node.Value, " ")
}

// PrintReverse печатает список с конца.
func (fl *Flist) PrintReverse() {
        fl.printReverseFrom(fl.head)
        fmt.Println()
}

// PrintList печатает список с начала.
func (fl *Flist) PrintList() {
        for p := fl.head; p != nil; p = p.Next {
                fmt.Print(p.Value, " -> ")
        }
        fmt.Println("nil")
}

// GetHead возвращает первый узел списка.
func (fl *Flist) GetHead() *FNode {
        return fl.head
}

// HandleFlist выполняет команду над односвязным списком с именем name.

                node = fl.FindValue(target)
                if node == nil {
                        fmt.Println("нет такого элемента")
                } else {
                        fl.DeleteBefore(node)
                }
        case FDEL:
                fmt.Fscan(ss, &value)
                fl.DeleteValue(value)
        case FFIND:
                fmt.Fscan(ss, &value)
                if fl.FindValue(value) != nil {
                        fmt.Println("-> TRUE")
                } else {
                        fmt.Println("-> FALSE")
                }
        case FPRINT:
                fl.PrintList()
        case FPRINTR:
                fl.PrintReverse()
        }

        items := []string{}
        for current := fl.GetHead(); current != nil; current = current.Next {
                items = append(items, current.Value)
        }
        saveStrings(db, name, "flist", items)
}

// foldSum - хеш методом свертки: число режется на куски по 3 цифры,
// куски складываются.
func foldSum(key int) int {
        if key < 0 {
                key = -key
        }
        s := strconv.Itoa(key)
        sum := 0
        for i := 0; i < len(s); i += 3 {
                end := i + 3
                if end > len(s) {
                        end = len(s)
                }
                part, _ := strconv.Atoi(s[i:end])
                sum += part
        }
        return sum
}

const hashSize = 100

// Node - узел цепочки в хеш-таблице.
type Node struct {
        Value int
        Next  *Node
}

// HashTable - хеш-таблица с цепочками.
type HashTable struct {
        table [hashSize]*Node
}

// NewHashTable создает пустую таблицу.
func NewHashTable() *HashTable {
        return &HashTable{}
}

func (ht *HashTable) hashKey(key int) int {
        return foldSum(key) % hashSize
}

// Search проверяет, есть ли ключ в таблице.
func (ht *HashTable) Search(key int) bool {
        for current := ht.table[ht.hashKey(key)]; current != nil; current = current.Next {
                if current.Value == key {
                        return true
                }
        }
        return false
}

// Add добавляет ключ в начало цепочки своей ячейки. Повторы не добавляются.
func (ht *HashTable) Add(key int) {
        if ht.Search(key) {
                return
        }
        index := ht.hashKey(key)
        ht.table[index] = &Node{Value: key, Next: ht.table[index]}
}

// Remove удаляет ключ из таблицы.
func (ht *HashTable) Remove(key int) {
        index := ht.hashKey(key)

        var prev *Node
        for current := ht.table[index]; current != nil; current = current.Next {
                if current.Value == key {
                        if prev == nil {
                                ht.table[index] = current.Next
                        } else {
                                prev.Next = current.Next
                        }
                        return
                }
                prev = current
        }
}

// Print печатает непустые ячейки с их цепочками.
func (ht *HashTable) Print() {
        for i := 0; i < hashSize; i++ {
                if ht.table[i] == nil {
                        continue
                }
                fmt.Print(i, ": ")
                for current := ht.table[i]; current != nil; current = current.Next {
                        fmt.Print(current.Value, " -> ")
                }
                fmt.Println("nil")
        }
}

// GetAll возвращает все ключи таблицы.
func (ht *HashTable) GetAll() []int {
        result := []int{}
        for i := 0; i < hashSize; i++ {
                for current := ht.table[i]; current != nil; current = current.Next {
                        result = append(result, current.Value)
                }
        }
        return result
}

const maxTempt = 50

// Kukushka - хеш-таблица кукушки: две таблицы и две хеш-функции.
type Kukushka struct {
        size  int
        t1    []int
        t2    []int
        used1 []bool
        used2 []bool
}

// NewKukushka создает пустую таблицу на 11 ячеек.
func NewKukushka() *Kukushka {
        size := 11
        return &Kukushka{
                size:  size,
                t1:    make([]int, size),
                t2:    make([]int, size),
                used1: make([]bool, size),
                used2: make([]bool, size),
        }
}

func (k *Kukushka) hash1(key int) int {
        return foldSum(key) % k.size
}

func (k *Kukushka) hash2(key int) int {
        h := key % k.size
        if h < 0 {
                h += k.size
        }
        return h
}

// moreplace увеличивает таблицы в два раза и заново вставляет все ключи.
func (k *Kukushka) moreplace() {
        oldSize := k.size
        old1, old2 := k.t1, k.t2
        oldUsed1, oldUsed2 := k.used1, k.used2

        k.size *= 2
        k.t1 = make([]int, k.size)
        k.t2 = make([]int, k.size)
        k.used1 = make([]bool, k.size)
        k.used2 = make([]bool, k.size)

        for i := 0; i < oldSize; i++ {
                if oldUsed1[i] {
                        k.Insert(old1[i])
                }
                if oldUsed2[i] {
                        k.Insert(old2[i])
                }
        }
}

// Search проверяет, есть ли ключ в одной из двух таблиц.
func (k *Kukushka) Search(key int) bool {
        i1 := k.hash1(key)
        if k.used1[i1] && k.t1[i1] == key {
                return true
        }
        i2 := k.hash2(key)
        return k.used2[i2] && k.t2[i2] == key
}

// Insert вставляет ключ. Если место занято, старый ключ выталкивается
// в другую таблицу. Если вытеснения идут по кругу, таблицы расширяются.
func (k *Kukushka) Insert(key int) {
        if k.Search(key) {
                return
        }

        cur := key
        for tempt := 0; tempt < maxTempt; tempt++ {
                i1 := k.hash1(cur)
                if !k.used1[i1] {
                        k.t1[i1] = cur
                        k.used1[i1] = true
                        return
                }
                k.t1[i1], cur = cur, k.t1[i1]

                i2 := k.hash2(cur)
                if !k.used2[i2] {
                        k.t2[i2] = cur
                        k.used2[i2] = true
                        return
                }
                k.t2[i2], cur = cur, k.t2[i2]
        }
        k.moreplace()
        k.Insert(cur)
}

// Remove удаляет ключ из таблицы.
func (k *Kukushka) Remove(key int) {
        i1 := k.hash1(key)
        if k.used1[i1] && k.t1[i1] == key {
                k.used1[i1] = false
                return
        }
        i2 := k.hash2(key)
        if k.used2[i2] && k.t2[i2] == key {
                k.used2[i2] = false
        }
}

// Print печатает занятые ячейки обеих таблиц.
func (k *Kukushka) Print() {
        fmt.Print("t1: ")
        for i := 0; i < k.size; i++ {

                        fmt.Printf("[%d]=%d ", i, k.t2[i])
                }
        }
        fmt.Println()
}

        result := []int{}

// QNode - узел очереди.
type QNode struct {
        A    string
        Next *QNode
}

// Queue - очередь на односвязном списке: забираем из head, добавляем в tail.
type Queue struct {
        head *QNode
        tail *QNode
}

// NewQueue создает пустую очередь.
func NewQueue() *Queue {
        return &Queue{}
}

// IsEmptyQ проверяет, пуста ли очередь.
func (q *Queue) IsEmptyQ() bool {
        return q.head == nil
}

// Push добавляет элемент в конец очереди.
func (q *Queue) Push(a string) {
        newNode := &QNode{A: a}

        if q.IsEmptyQ() {
                q.head = newNode
                q.tail = newNode
        } else {
                q.tail.Next = newNode
                q.tail = newNode
        }
}

// Pop убирает первый элемент очереди.
func (q *Queue) Pop() {
        if q.IsEmptyQ() {
                return
        }

        q.head = q.head.Next
        if q.IsEmptyQ() {
                q.tail = nil
        }
}

// Top возвращает первый элемент, для пустой очереди - пустую строку.
func (q *Queue) Top() string {
        if q.IsEmptyQ() {
                return ""
        }
        return q.head.A
}

// PrintQ печатает очередь от начала к концу.
func (q *Queue) PrintQ() {
        for current := q.head; current != nil; current = current.Next {
                fmt.Print(current.A, " -> ")
        }
        fmt.Println("nil")
}

// GetHead возвращает первый узел очереди.
func (q *Queue) GetHead() *QNode {
        return q.head
}

        case QPOP:
                if q.IsEmptyQ() {
                        fmt.Println("-> очередь пуста")
                } else {
                        fmt.Println("->", q.Top())
                        q.Pop()
                }
        case QPRINT:
                q.PrintQ()
        }

        items := []string{}
        for current := q.GetHead(); current != nil; current = current.Next {

        }

// Общие функции для бинарной сериализации.
// Число пишется как 4 байта, строка - как длина (число) и сами байты.

func writeInt(out io.Writer, value int) {
        _ = binary.Write(out, binary.LittleEndian, int32(value))
}

func readInt(in io.Reader) int {
        var value int32
        _ = binary.Read(in, binary.LittleEndian, &value)
        return int(value)
}

func writeString(out io.Writer, text string) {
        writeInt(out, len(text))
        _, _ = out.Write([]byte(text))
}

func readString(in io.Reader) string {
        length := readInt(in)
        buffer := make([]byte, length)
        _, _ = io.ReadFull(in, buffer)
        return string(buffer)
}

import (

// TreeNode - узел двоичного дерева поиска.
type TreeNode struct {
        Key   int
        Left  *TreeNode
        Right *TreeNode
}

// Tree - двоичное дерево поиска.
type Tree struct {
        root *TreeNode
}

// NewTree создает пустое дерево.
func NewTree() *Tree {
        return &Tree{}
}

// Insert вставляет ключ: меньшие уходят влево, остальные вправо.
func (tr *Tree) Insert(key int) {
        newNode := &TreeNode{Key: key}
        if tr.root == nil {
                tr.root = newNode
                return
        }

        var parent *TreeNode
        current := tr.root
        for current != nil {
                parent = current
                if key < current.Key {
                        current = current.Left
                } else {
                        current = current.Right
                }
        }

        if key < parent.Key {
                parent.Left = newNode
        } else {
                parent.Right = newNode
        }
}

// Рекурсивные обходы. Им нужен узел, с которого начинать,
// поэтому снаружи они вызываются через PrintPreorder и т.д.

func (tr *Tree) preorder(node *TreeNode) {
        if node == nil {
                return
        }
        fmt.Print(node.Key, " ")
        tr.preorder(node.Left)
        tr.preorder(node.Right)
}

func (tr *Tree) inorder(node *TreeNode) {
        if node == nil {
                return
        }
        tr.inorder(node.Left)
        fmt.Print(node.Key, " ")
        tr.inorder(node.Right)
}

func (tr *Tree) postorder(node *TreeNode) {
        if node == nil {
                return
        }
        tr.postorder(node.Left)
        tr.postorder(node.Right)
        fmt.Print(node.Key, " ")
}

// PrintPreorder печатает прямой обход.
func (tr *Tree) PrintPreorder() {
        tr.preorder(tr.root)
}

// PrintInorder печатает симметричный обход.
func (tr *Tree) PrintInorder() {
        tr.inorder(tr.root)
}

// PrintPostorder печатает обратный обход.
func (tr *Tree) PrintPostorder() {
        tr.postorder(tr.root)
}

// Bfs печатает обход в ширину.
func (tr *Tree) Bfs() {
        if tr.root == nil {
                return
        }

        ochered := []*TreeNode{tr.root}
        for i := 0; i < len(ochered); i++ {
                node := ochered[i]
                fmt.Print(node.Key, " ")
                if node.Left != nil {
                        ochered = append(ochered, node.Left)
                }
                if node.Right != nil {
                        ochered = append(ochered, node.Right)
                }
        }
}

// IsComplete проверяет, что дерево полное: при обходе в ширину
// после первого пустого места не должно встречаться узлов.
func (tr *Tree) IsComplete() bool {
        if tr.root == nil {
                return true
        }

        ochered := []*TreeNode{tr.root}
        sawEmpty := false
        for i := 0; i < len(ochered); i++ {
                node := ochered[i]
                if node == nil {
                        sawEmpty = true
                } else {
                        if sawEmpty {
                                return false
                        }
                        ochered = append(ochered, node.Left, node.Right)
                }
        }
        return true
}

// Search ищет узел с ключом key, если не нашел - возвращает nil.
func (tr *Tree) Search(key int) *TreeNode {
        current := tr.root
        for current != nil {
                if key == current.Key {
                        return current
                }
                if key < current.Key {
                        current = current.Left
                } else {
                        current = current.Right
                }
        }
        return nil
}

// GetRoot возвращает корень дерева.
func (tr *Tree) GetRoot() *TreeNode {
        return tr.root
}

// bfsKeys возвращает ключи в порядке обхода в ширину.
// Если вставить их в том же порядке, получится такое же дерево.
func (tr *Tree) bfsKeys() []int {
        keys := []int{}
        if tr.root == nil {
                return keys
        }

        ochered := []*TreeNode{tr.root}
        for i := 0; i < len(ochered); i++ {
                node := ochered[i]
                keys = append(keys, node.Key)
                if node.Left != nil {
                        ochered = append(ochered, node.Left)
                }
                if node.Right != nil {
                        ochered = append(ochered, node.Right)
                }
        }
        return keys
}

// HandleTree выполняет команду над деревом с именем name.
func HandleTree(command Command, db DB, name string, ss io.Reader) {
        if WrongType(db, name, "tree") {
                return
        }

        tr := NewTree()
        for _, key := range loadInts(db, name) {
                tr.Insert(key)
        }

        var value int
        switch command {
        case TINSERT:
                fmt.Fscan(ss, &value)
                tr.Insert(value)
                fmt.Println("->", value)
        case TFIND:
                fmt.Fscan(ss, &value)
                if tr.Search(value) != nil {
                        fmt.Println("-> TRUE")
                } else {
                        fmt.Println("-> FALSE")
                }
        case TCOMPLETE:
                if tr.IsComplete() {
                        fmt.Println("-> TRUE")
                } else {
                        fmt.Println("-> FALSE")
                }
        case TPRINT:
                fmt.Print("прямой:       ")
                tr.PrintPreorder()
                fmt.Println()
                fmt.Print("симметричный: ")
                tr.PrintInorder()
                fmt.Println()
                fmt.Print("обратный:     ")
                tr.PrintPostorder()
                fmt.Println()
                fmt.Print("в ширину:     ")
                tr.Bfs()
                fmt.Println()
        }

        saveInts(db, name, "tree", tr.bfsKeys())
}

// SaveBinary сохраняет дерево в бинарный файл:
// количество ключей, затем ключи в порядке обхода в ширину.
func (tr *Tree) SaveBinary(fileName string) {
        out, err := os.Create(fileName)
        if err != nil {
                return
        }
        defer out.Close()

        keys := tr.bfsKeys()
        writeInt(out, len(keys))
        for _, key := range keys {
                writeInt(out, key)
        }
}

// LoadBinary загружает дерево из бинарного файла.
// Старые ключи перед загрузкой удаляются.
func (tr *Tree) LoadBinary(fileName string) {
        in, err := os.Open(fileName)
        if err != nil {
                return
        }
        defer in.Close()

        tr.root = nil
        count := readInt(in)
        for i := 0; i < count; i++ {
                tr.Insert(readInt(in))
        }
}
